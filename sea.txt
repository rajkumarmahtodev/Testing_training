package pages;

import org.openqa.selenium.By;
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.Keys;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

import java.time.Duration;
import java.util.List;

public class SearchPage {

    private WebDriver driver;
    private WebDriverWait wait;

    public SearchPage(WebDriver driver) {

        this.driver = driver;

        this.wait = new WebDriverWait(
                driver,
                Duration.ofSeconds(15)
        );
    }

    // Search box
    private By searchInput = By.xpath(
            "//input[contains(@placeholder,'Search') or contains(@placeholder,'search')]"
    );

    // Filters
    private By vegText = By.xpath(
            "//*[normalize-space()='Veg']"
    );

    private By nonVegText = By.xpath(
            "//*[normalize-space()='Non-Veg']"
    );

    // Product elements
    private By productNames = By.xpath(
            "//h3"
    );

    private By productLinks = By.xpath(
            "//a[contains(@href,'menu')]"
    );


    public void openSearchPage() {

        driver.get(
                "https://www.mcdelivery.co.in/search"
        );

        wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                        searchInput
                )
        );
    }


    public boolean isSearchBoxDisplayed() {

        try {

            return wait.until(
                    ExpectedConditions.visibilityOfElementLocated(
                            searchInput
                    )
            ).isDisplayed();

        } catch (Exception e) {

            return false;
        }
    }


    public void clickSearchBox() {

        wait.until(
                ExpectedConditions.elementToBeClickable(
                        searchInput
                )
        ).click();
    }


    public void enterSearchKeyword(String keyword) {

        WebElement search = wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                        searchInput
                )
        );

        search.clear();

        search.sendKeys(keyword);

        search.sendKeys(Keys.ENTER);

        waitForPageLoad();
    }


    public void typeSearchKeyword(String keyword) {

        WebElement search = wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                        searchInput
                )
        );

        search.clear();

        search.sendKeys(keyword);
    }


    public String getSearchValue() {

        return wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                        searchInput
                )
        ).getAttribute("value");
    }


    public void clearSearch() {

        WebElement search = wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                        searchInput
                )
        );

        search.clear();

        search.sendKeys(Keys.ENTER);

        waitForPageLoad();
    }


    public boolean isSearchResultDisplayed(
            String keyword) {

        String pageText =
                driver.findElement(
                        By.tagName("body")
                )
                .getText()
                .toLowerCase();

        return pageText.contains(
                keyword.toLowerCase()
        );
    }


    public int getNumberOfProducts() {

        List<WebElement> products =
                driver.findElements(productNames);

        return products.size();
    }


    public boolean hasProducts() {

        return getNumberOfProducts() > 0;
    }


    public boolean isNoResultsMessageDisplayed() {

        String pageText =
                driver.findElement(
                        By.tagName("body")
                )
                .getText()
                .toLowerCase();

        return pageText.contains("no results")
                || pageText.contains("no food")
                || pageText.contains("not found")
                || pageText.contains("no items");
    }


    public String getCurrentUrl() {

        return driver.getCurrentUrl();
    }


    public boolean urlContains(String text) {

        return driver.getCurrentUrl()
                .toLowerCase()
                .contains(
                        text.toLowerCase()
                );
    }


    public boolean isVegFilterDisplayed() {

        try {

            return wait.until(
                    ExpectedConditions.visibilityOfElementLocated(
                            vegText
                    )
            ).isDisplayed();

        } catch (Exception e) {

            return false;
        }
    }


    public void clickVegFilter() {

        WebElement veg = wait.until(
                ExpectedConditions.elementToBeClickable(
                        vegText
                )
        );

        veg.click();

        waitForPageLoad();
    }


    public boolean isNonVegFilterDisplayed() {

        try {

            return wait.until(
                    ExpectedConditions.visibilityOfElementLocated(
                            nonVegText
                    )
            ).isDisplayed();

        } catch (Exception e) {

            return false;
        }
    }


    public void clickNonVegFilter() {

        WebElement nonVeg = wait.until(
                ExpectedConditions.elementToBeClickable(
                        nonVegText
                )
        );

        nonVeg.click();

        waitForPageLoad();
    }


    public void clickFirstProduct() {

        List<WebElement> products =
                wait.until(
                        ExpectedConditions
                                .presenceOfAllElementsLocatedBy(
                                        productLinks
                                )
                );

        if (!products.isEmpty()) {

            WebElement firstProduct =
                    products.get(0);

            ((JavascriptExecutor) driver)
                    .executeScript(
                            "arguments[0].scrollIntoView(true);",
                            firstProduct
                    );

            firstProduct.click();

            waitForPageLoad();
        }
    }


    public boolean isProductDetailsPageDisplayed() {

        String url =
                driver.getCurrentUrl();

        String pageText =
                driver.findElement(
                        By.tagName("body")
                ).getText();

        return !url.equals(
                "https://www.mcdelivery.co.in/"
        )
                && pageText.length() > 50;
    }


    private void waitForPageLoad() {

        try {

            wait.until(driver ->
                    ((JavascriptExecutor) driver)
                            .executeScript(
                                    "return document.readyState"
                            )
                            .equals("complete")
            );

        } catch (Exception ignored) {

        }
    }
}
