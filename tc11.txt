
public boolean isNoResultsMessageDisplayed() {
    By locator = By.cssSelector("span.empty-text");

    try {
        WebElement message = wait.until(
                ExpectedConditions.visibilityOfElementLocated(locator)
        );

        String text = message.getText();

        System.out.println("DEBUG: No-results text = " + text);
        System.out.println("DEBUG: Current URL = " + driver.getCurrentUrl());

        return message.isDisplayed()
                && text.toLowerCase().contains("no results found");

    } catch (Exception e) {
        System.out.println("DEBUG: Current URL = " + driver.getCurrentUrl());

        System.out.println(
                "DEBUG: empty-text elements found = "
                + driver.findElements(locator).size()
        );

        try {
            String bodyText = driver.findElement(By.tagName("body")).getText();

            System.out.println(
                    "DEBUG: Page text = "
                    + bodyText.substring(0, Math.min(bodyText.length(), 1500))
            );
        } catch (Exception ex) {
            System.out.println("DEBUG: Could not read page text.");
        }

        e.printStackTrace();
        return false;
    }
}
